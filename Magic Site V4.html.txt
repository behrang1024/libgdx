        // Functions
        function searchCards() {
            let query = searchInput.value.trim();
            let colorQuery = selectedColors.length > 0 ? `color=${selectedColors.join('')}` : '';
            let typeQuery = typeFilter.value ? `type=${typeFilter.value}` : '';
            let rarityQuery = rarityFilter.value ? `rarity=${rarityFilter.value}` : '';

            // Build CMC query
            let cmcQuery = '';
            if (selectedCMC.length > 0) {
                const cmcConditions = selectedCMC.map(cmc => {
                    if (cmc === '7') {
                        return 'cmc>=7';
                    } else {
                        return `cmc=${cmc}`;
                    }
                });
                cmcQuery = cmcConditions.join(' or ');
            }

            let url = `https://api.scryfall.com/cards/search?q=${encodeURIComponent(query)}`;

            if (colorQuery) url += `+${colorQuery}`;
            if (typeQuery) url += `+${typeQuery}`;
            if (rarityQuery) url += `+${rarityQuery}`;
            if (cmcQuery) url += `+(${cmcQuery})`;

            url += `&page=${currentPage}`;

            fetch(url)
                .then(response => response.json())
                .then(data => {
                    if (data.object === 'error') {
                        searchResults.innerHTML = `
                            <div class="text-center text-gray-500 py-10 col-span-full">
                                <i class="fas fa-exclamation-circle fa-2x mb-2"></i>
                                <p>${data.details}</p>
                            </div>
                        `;
                        prevPageBtn.disabled = true;
                        nextPageBtn.disabled = true;
                        return;
                    }

                    totalPages = data.total_cards ? Math.ceil(data.total_cards / data.per_page) : 1;
                    currentPageSpan.textContent = currentPage;
                    prevPageBtn.disabled = currentPage <= 1;
                    nextPageBtn.disabled = currentPage >= totalPages;

                    displaySearchResults(data.data);
                })
                .catch(error => {
                    console.error('Error:', error);
                    searchResults.innerHTML = `
                        <div class="text-center text-gray-500 py-10 col-span-full">
                            <i class="fas fa-exclamation-triangle fa-2x mb-2"></i>
                            <p>Error loading cards. Please try again.</p>
                        </div>
                    `;
                    prevPageBtn.disabled = true;
                    nextPageBtn.disabled = true;
                });
        }

        function renderManaCost(manaCost) {
            if (!manaCost) return '';

            const symbols = manaCost.match(/\{[^}]+\}/g) || [];
            return symbols.map(symbol => {
                const cost = symbol.replace(/[{}]/g, '').toLowerCase();
                let colorClass = '';

                switch(cost) {
                    case 'w': colorClass = 'bg-white text-black'; break;
                    case 'u': colorClass = 'bg-blue-600 text-white'; break;
                    case 'b': colorClass = 'bg-black text-white'; break;
                    case 'r': colorClass = 'bg-red-600 text-white'; break;
                    case 'g': colorClass = 'bg-green-600 text-white'; break;
                    default: colorClass = 'bg-gray-400 text-black'; break;
                }

                return `<span class="inline-flex items-center justify-center w-5 h-5 rounded-full text-xs font-bold ${colorClass} mx-px">${cost.toUpperCase()}</span>`;
            }).join('');
        }

        function addToDeck(card, quantity) {
            // Check if card is already in deck
            const existingCardIndex = currentDeck.findIndex(c => c.id === card.id);

            if (existingCardIndex >= 0) {
                // Update quantity
                currentDeck[existingCardIndex].quantity += quantity;

                // Basic lands can have any number, other cards max 4
                if (!card.type_line.toLowerCase().includes('basic land') && currentDeck[existingCardIndex].quantity > 4) {
                    currentDeck[existingCardIndex].quantity = 4;
                    alert('You can only have 4 copies of a non-basic land card in your deck.');
                }
            } else {
                // Add new card
                currentDeck.push({
                    id: card.id,
                    name: card.name,
                    image: card.image_uris ? card.image_uris.normal : card.card_faces[0].image_uris.normal,
                    type: card.type_line,
                    mana_cost: card.mana_cost,
                    colors: card.colors || [],
                    cmc: card.cmc || 0,
                    quantity: Math.min(quantity, card.type_line.toLowerCase().includes('basic land') ? quantity : 4)
                });                
            }

            updateDeckDisplay();
            cardModal.classList.add('hidden');
        }

        function updateDeckStats() {
            // Count card types
            let creatures = 0;
            let spells = 0;
            let lands = 0;

            currentDeck.forEach(card => {
                if (card.type.toLowerCase().includes('creature')) {
                    creatures += card.quantity;
                } else if (card.type.toLowerCase().includes('land')) {
                    lands += card.quantity;
                } else {
                    spells += card.quantity;
                }
            });

            creatureCount.textContent = creatures;
            spellCount.textContent = spells;
            landCount.textContent = lands;

            // Update color distribution
            const colorCounts = {
                W: 0, // White
                U: 0, // Blue
                B: 0, // Black
                R: 0, // Red
                G: 0, // Green
                C: 0  // Colorless
            };

            currentDeck.forEach(card => {
                if (card.colors && card.colors.length > 0) {
                    card.colors.forEach(color => {
                        colorCounts[color] += card.quantity;
                    });
                } else if (!card.type.toLowerCase().includes('land')) {
                    colorCounts.C += card.quantity;
                } 
            });

            // Add colors from commander
            currentCommander.forEach(card => {
                if (card.colors && card.colors.length > 0) {
                    card.colors.forEach(color => {
                        colorCounts[color] += card.quantity;
                    });
                } else if (!card.type.toLowerCase().includes('land')) {
                    colorCounts.C += card.quantity;
                } 
            });

            colorPieChart.data.datasets[0].data = [
                colorCounts.W,
                colorCounts.U,
                colorCounts.B,
                colorCounts.R,
                colorCounts.G,
                colorCounts.C
            ];
            colorPieChart.update();
            
            

            // Update mana curve
            const curve = [0, 0, 0, 0, 0, 0, 0]; // 0, 1, 2, 3, 4, 5, 6+

            const allCards = [...currentDeck, ...currentCommander];
            
            allCards.forEach(card => {
            if (!card.type.toLowerCase().includes('land')) {
                let cmc = Math.min(6, Math.floor(card.cmc || 0));
                for (let i = 0; i < card.quantity; i++) {
                    curve[cmc]++;
                }
            }
        });
            manaCurveChart.data.datasets[0].data = curve;
            manaCurveChart.update();
        }

        function loadSharedDeck(encodedDeck) {
            try {
                const deckData = JSON.parse(atob(encodedDeck));

                // Clear current deck and sideboard
                currentDeck = [];
                currentSideboard = [];
                deckName.value = deckData.name;
                deckFormat.value = deckData.format || 'commander';
                //commanderSection.classList.remove('hidden');
                

                // For each card in the shared deck, we need to fetch its details
                // This is a simplified version - in a real app you'd want better error handling
                const fetchPromises = deckData.cards.map(card => {
                    return fetch(`https://api.scryfall.com/cards/${card.id}`)
                        .then(response => response.json())
                        .then(cardData => {
                            currentDeck.push({
                                id: cardData.id,
                                name: cardData.name,
                                image: cardData.image_uris ? cardData.image_uris.normal : cardData.card_faces[0].image_uris.normal,
                                type: cardData.type_line,
                                mana_cost: cardData.mana_cost,
                                colors: cardData.colors || [],
                                cmc: cardData.cmc || 0,
                                quantity: card.quantity
                            });
                        })
                        .catch(error => {
                            console.error('Error loading card:', card.id, error);
                        });
                });

                // For each card in the shared sideboard, fetch its details
                const sideboardPromises = deckData.sideboard.map(card => {
                    return fetch(`https://api.scryfall.com/cards/${card.id}`)
                        .then(response => response.json())
                        .then(cardData => {
                            currentSideboard.push({
                                id: cardData.id,
                                name: cardData.name,
                                image: cardData.image_uris ? cardData.image_uris.normal : cardData.card_faces[0].image_uris.normal,
                                type: cardData.type_line,
                                mana_cost: cardData.mana_cost,
                                colors: cardData.colors || [],
                                cmc: cardData.cmc || 0,
                                quantity: card.quantity
                            });
                        })
                        .catch(error => {
                            console.error('Error loading sideboard card:', card.id, error);
                        });
                });

                Promise.all([...fetchPromises, ...sideboardPromises]).then(() => {
                    updateDeckDisplay();
                    updateSideboardDisplay();
                });
                updateCommanderDisplay();

            } catch (error) {
                console.error('Error loading shared deck:', error);
                alert('Error loading shared deck. Creating a new deck instead.');
                newDeck();
            }
        }

        function handleLogin() {
            console.log('Attempting login...');
            const email = loginEmail.value.trim();
            const password = loginPassword.value.trim();

            if (!email || !password) {
                loginError.textContent = 'Please enter both email and password';
                loginError.classList.remove('hidden');
                return;
            }

            fetch('/login', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify({ email, password }),
            })
            .then(response => response.json().then(data => ({ status: response.status, body: data })))
            .then(({ status, body }) => {
                if (status === 200) { // Success
                    currentUser = { id: body.user_id, username: body.username, email: email };
                    localStorage.setItem('mtgCurrentUser', JSON.stringify(currentUser));
                    loginError.classList.add('hidden');
                    loginForm.classList.add('hidden');
                    loggedInUsername.textContent = currentUser.username;
                    loggedInView.classList.remove('hidden');
                    loginEmail.value = '';
                    loginPassword.value = '';
                    myDecksBtn.classList.remove('hidden');
                    myDecksBtn.classList.add('inline-flex');
                } else {
                    loginError.textContent = body.error || 'Login failed';
                    loginError.classList.remove('hidden');
                }
            })
            .catch(error => {
                console.error('Login error:', error);
                loginError.textContent = 'Login failed due to a network error';
                loginError.classList.remove('hidden');
            });
        }

            const email = registerEmail.value;

            // Send registration data to the backend
            fetch('/register', {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify({ username, email, password }),
            })
            .then(response => {
                if (response.ok) {
                    return response.json();
                } else {
                    return response.json().then(err => { throw err; });
                }
            })
            .then(data => {
                alert('User registered successfully');

                registerPassword.value = '';

        /**
         * Fetches and loads a specific deck based on its deck ID.
         * This function retrieves the deck's details from the server and then populates
         * the currentDeck and currentSideboard arrays with the deck's card data.
         * It updates the UI to reflect the loaded deck's name, format, and contents.
         */
        function loadDeck(deckId) {
            console.log("loadDeck function called with deckId:", deckId);
            fetch(`/get_deck?deck_id=${deckId}`)
    .then(response => response.json())
    .then(deck => {
        console.log("Deck data received:", deck);


        if (!deck) {
            alert("Deck not found!");
            return;
        }

        // Clear existing deck and sideboard
        currentDeck = [];
        currentSideboard = [];

        deckName.value = deck.deck_name; // Set the deck name
        deckFormat.value = deck.format; // Set the deck format

        // Function to fetch and process each card
        const processCard = (card, targetList) => {
            return fetch(`https://api.scryfall.com/cards/search?q=${encodeURIComponent(card.card_name)}`)
                .then(response => {
                    if (!response.ok) {
                        throw new Error(`HTTP error! status: ${response.status}`);
                    }
                    return response.json();
                })
                .then(data => {
                    if (data.data && data.data.length > 0) {
                        const cardData = data.data[0];
                        targetList.push({
                            id: cardData.id,
                            name: cardData.name,
                            image: cardData.image_uris ? cardData.image_uris.normal : (cardData.card_faces ? cardData.card_faces[0].image_uris.normal : ''),
                            type: cardData.type_line,
                            mana_cost: cardData.mana_cost,
                            colors: cardData.colors || [],
                            cmc: cardData.cmc,
                            quantity: card.quantity
                        });
                    } else {
                        console.error("Card not found:", card.card_name);
                    }
                })
                .catch(error => {
                    console.error("Error loading card:", card.card_name, error);
                });
        };

        // Create an array of promises for fetching and processing each mainboard card
        const mainboardPromises = deck.cards.map(card => processCard(card, currentDeck));

        // Create an array of promises for fetching and processing each sideboard card
        const sideboardPromises = deck.sideboard.map(card => processCard(card, currentSideboard));

        // Wait for all promises to resolve
        Promise.all([...mainboardPromises, ...sideboardPromises])
            .then(() => {
                updateDeckDisplay(); // Update the display of the deck
                updateSideboardDisplay(); // Update the display of the sideboard
            })
            .catch(error => {
                console.error("Error processing cards:", error);
                alert("Failed to process some cards in the deck.");
            });
        updateCommanderDisplay();

    })
    .catch(error => {
        console.error("Error loading deck:", error);
        alert("Failed to load deck.");
    });
}
